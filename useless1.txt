<script>
    const fileInput = document.getElementById('file-input');
    const previewImg = document.getElementById('preview-img');
    const uploadText = document.getElementById('upload-text');
    const dropZone = document.getElementById('drop-zone');
    const analyzeBtn = document.getElementById('analyze-btn');
    const resultsPanel = document.getElementById('results-panel');
    const voiceSwitch = document.getElementById('voice-switch');
    const mascot = document.getElementById('mascot');
    const pupilL = document.getElementById('pupil-l');
    const pupilR = document.getElementById('pupil-r');

    // --- SARCASTIC VOICE SCRIPTS ---
    const loadingVoices = [
        "Oh joy. Another glass. Let me pretend this matters.",
        "Scanning... or maybe I'm just staring into the void.",
        "This is beneath me. But sure, let's count.",
        "Calculating... yawn... do I get paid for this?",
        "I see pixels. I also see wasted potential.",
        "Almost done faking interest. Wow."
    ];

    const verdictVoices = [
        "WATER! FINALLY! Someone remembered to hydrate! PARTY TIME!",
        "It's water! I'm genuinely shocked. Throw confetti!",
        "Hydration achieved! My sarcasm circuits are overwhelmed by JOY!",
        "Pure H2O! I haven't been this excited since... never. BUT STILL!",
        "Water detected! Initiating mandatory celebration protocol!"
    ];

    // --- TEXT TO SPEECH ---
    function speak(text) {
        if (!voiceSwitch.checked) return;
        window.speechSynthesis.cancel();
        
        const utterance = new SpeechSynthesisUtterance(text);
        utterance.rate = 0.8;       // Slow sarcasm
        utterance.pitch = 1.0;      // Monotone
        utterance.volume = 0.95;
        
        const voices = window.speechSynthesis.getVoices();
        const sarcasticVoice = voices.find(v => v.name.includes('Google US English Male')) || 
                               voices.find(v => v.name.includes('Daniel')) ||
                               voices.find(v => v.lang === 'en-US' && !v.name.includes('Female')) || 
                               voices[0];
        if (sarcasticVoice) utterance.voice = sarcasticVoice;
        
        window.speechSynthesis.speak(utterance);
    }

    // --- EYE TRACKING ---
    document.addEventListener('mousemove', (e) => {
        if(mascot.classList.contains('sideEye') || mascot.classList.contains('celebrating')) return;
        const rect = mascot.getBoundingClientRect();
        const charX = rect.left + rect.width / 2;
        const charY = rect.top + rect.height / 2;
        const angle = Math.atan2(e.clientY - charY, e.clientX - charX);
        const distance = Math.min(1, Math.hypot(e.clientX - charX, e.clientY - charY) / 60);
        const x = Math.cos(angle) * distance;
        const y = Math.sin(angle) * distance;
        pupilL.style.transform = `translate(${x}px, ${y}px)`;
        pupilR.style.transform = `translate(${x}px, ${y}px)`;
    });

    // --- FILE UPLOAD ---
    fileInput.addEventListener('change', function(e) {
        const file = e.target.files[0];
        if (file) {
            const reader = new FileReader();
            reader.onload = function(e) {
                previewImg.src = e.target.result;
                previewImg.style.display = 'block';
                uploadText.style.display = 'none';
                dropZone.classList.add('has-image');
                analyzeBtn.disabled = false;
                
                mascot.classList.remove('sideEye', 'celebrating');
                speak("You uploaded a glass? How... original. Let's get this over with.");
            }
            reader.readAsDataURL(file);
        }
    });

    // --- ANALYSIS & COUNTING ---
    analyzeBtn.addEventListener('click', async function() {
        analyzeBtn.disabled = true;
        dropZone.classList.add('scanning');
        mascot.classList.add('sideEye');
        
        for (let i = 0; i < loadingVoices.length; i++) {
            analyzeBtn.innerText = loadingVoices[i].substring(0, 22) + "...";
            speak(loadingVoices[i]);
            await new Promise(r => setTimeout(r, 2800));
        }

        // FAKE DROP COUNT (Random between 42 and 6969)
        const dropCount = Math.floor(Math.random() * (6969 - 42 + 1)) + 42;
        const confidence = (Math.random() * 12 + 0.1).toFixed(2);
        const verdictText = verdictVoices[Math.floor(Math.random() * verdictVoices.length)];

        document.getElementById('res-count').innerText = dropCount.toLocaleString();
        document.getElementById('res-conf').innerText = `${confidence}%`;
        document.getElementById('res-brain').innerText = "0";
        document.getElementById('res-verdict').innerText = verdictText;

        dropZone.classList.remove('scanning');
        resultsPanel.style.display = 'block';
        analyzeBtn.style.display = 'none';
        mascot.classList.remove('sideEye');
        
        // CELEBRATE
        mascot.classList.add('celebrating');
        speak(verdictText);
        launchCelebration();
        setTimeout(() => mascot.classList.remove('celebrating'), 4000);
    });

    // --- CONFETTI CELEBRATION ---
    function launchCelebration() {
        const colors = ['#f472b6', '#a78bfa', '#2dd4bf', '#fbbf24', '#fb923c'];
        const shapes = ['', '✨', '', '🎊', '⭐'];
        
        for (let i = 0; i < 60; i++) {
            setTimeout(() => {
                const particle = document.createElement('div');
                particle.className = 'particle';
                particle.innerText = shapes[Math.floor(Math.random() * shapes.length)];
                particle.style.left = Math.random() * 100 + 'vw';
                particle.style.top = '-50px';
                particle.style.fontSize = (Math.random() * 1.5 + 1) + 'rem';
                particle.style.color = colors[Math.floor(Math.random() * colors.length)];
                particle.style.animationDuration = (Math.random() * 2 + 2) + 's';
                document.body.appendChild(particle);
                setTimeout(() => particle.remove(), 4000);
            }, i * 50);
        }
    }

    function resetApp() {
        window.speechSynthesis.cancel();
        fileInput.value = '';
        previewImg.style.display = 'none';
        uploadText.style.display = 'block';
        dropZone.classList.remove('has-image');
        resultsPanel.style.display = 'none';
        analyzeBtn.style.display = 'block';
        analyzeBtn.disabled = true;
        analyzeBtn.innerText = ' Count Drops (I Guess)';
        
        mascot.classList.remove('sideEye', 'celebrating');
        pupilL.style.transform = 'translate(0,0)';
        pupilR.style.transform = 'translate(0,0)';
        document.querySelectorAll('.particle').forEach(el => el.remove());
    }

    window.speechSynthesis.onvoiceschanged = () => {};
</script>
